# NS_Solver Experimental Cylinder Configuration
# This file is ignored unless enable_experimental_config is true.

# Master switch for every option in this file.

cylinder_free_x_damping = 5.0
cylinder_free_x_stiffness = 20.0

cylinder_free_x_max_displacement_percent = 25.0
cylinder_free_x_max_speed = 0.25
# Hold x fixed until this simulation time; 0 releases immediately.
cylinder_free_x_release_time = 0.0

cylinder_free_y_mass = 100.0
cylinder_free_y_damping = 5.0

cylinder_free_y_max_speed = 0.25
# Hold y fixed until this simulation time; 0 releases immediately.
